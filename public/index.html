Connect Your App or SDK

Point any client to your gateway URL. Keys auto-rotate transparently — your app doesn't need to know.

Python 
from openai import OpenAI

client = OpenAI(
    base_url="http://localhost:3000/v1",
    api_key="sk-switch-master-key"
)

# Keys auto-rotate on 429 — no changes needed
response = client.chat.completions.create(
    model="glm-5-3-flash",  # or "deepseek-v4-1-flash-1017" — see /v1/models
    messages=[{"role": "user", "content": "Hello!"}],
    stream=True
)
for chunk in response:
    if chunk.choices[0].delta.content:
        print(chunk.choices[0].delta.content, end="")
Node.js / TypeScript 
import OpenAI from 'openai';

const client = new OpenAI({
  baseURL: 'http://localhost:3000/v1',
  apiKey: 'sk-switch-master-key',
});

const res = await client.chat.completions.create({
  model: 'glm-5-3-flash',  // or 'deepseek-v4-1-flash-1017' — see /v1/models
  messages: [{ role: 'user', content: 'Hello!' }],
});
console.log(res.choices[0].message.content);
cURL

Anthropic SDK (Claude-compatible)

Use endpoint: http://localhost:3000

Anthropic messages route: POST /v1/messages

import anthropic

client = anthropic.Anthropic(
    base_url="http://localhost:3000",
    api_key="sk-switch-master-key"
)
msg = client.messages.create(
    model="glm-5-3-flash",  # or "deepseek-v4-1-flash-1017" — see /v1/models
    max_tokens=1024,
    messages=[{"role":"user","content":"Hello!"}]
)
print(msg.content[0].text)
